{{- /* Goldmark image render hook. Resolves from page bundle first, then assets/images/. Produces a WebP with the original as JPEG/PNG fallback. AVIF is omitted because Hugo's AVIF encoder hangs on some sources until the build times out. Keeps alt/title/attributes; applies lazy/async. Sets width and height so the browser reserves space before the image loads. Place at layouts/_markup/render-image.html to affect all Markdown images. */ -}} {{- $img := .Page.Resources.GetMatch (printf "%s*" (.Destination | path.Base)) -}} {{- if not $img -}} {{- $img = resources.Get (printf "images/%s" (.Destination | path.Base)) -}} {{- end -}} {{- if $img -}} {{- $webp := $img.Process "webp q85" -}} {{- else -}} {{- end -}}
