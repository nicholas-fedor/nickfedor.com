{{/* Homepage background. Processes assets/images/background.png (a 5120x1440 panorama) into WebP for the .home-background element. The background uses cover, so its height sets the scale and narrower viewports only show the center of the panorama. Narrower viewports therefore get center crops at full height, which render the same pixels as the full image while downloading less. Each crop is at least as wide as the visible area for its media query, with headroom for the navbar reducing the element's height. */}} {{ if .IsHome }} {{ $bg := resources.Get "images/background.png" }} {{ if $bg }} {{ $full := $bg.Process "webp q88" }} {{ $wide := $bg.Process (printf "crop 2560x%d Center webp q88" $bg.Height) }} {{ $narrow := $bg.Process (printf "crop 1800x%d Center webp q88" $bg.Height) }} {{ end }} {{ end }}
